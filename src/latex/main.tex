% ================= PÁGINA 2: SEÇÃO 1 (DNS) =================
\large{\textbf{\textcolor{AdguardGreen}{1. Bloquear Anúncios (DNS)}}}
\normalsize

\passo{AdguardGreen}{1}{Abra as \textbf{Configurações} do celular e vá em \textbf{Rede e Internet} (ou Conexões).}
\passo{AdguardGreen}{2}{Toque em \textbf{DNS Particular} e selecione "Nome do host do provedor".}
\passo{AdguardGreen}{3}{Digite \texttt{family.adguard-dns.com} e salve. (Isso bloqueia anúncios e conteúdo adulto).}

% ================= PÁGINA 3: SEÇÃO 2 (ECONOMIA GLOBAL) =================
\large{\textbf{\textcolor{BraveOrange}{2. Ativar a Economia de Dados}}}
\normalsize

\passo{BraveOrange}{1}{Nas \textbf{Configurações} do celular, toque na \textbf{Barra de Pesquisa} no topo da tela.}
\passo{BraveOrange}{2}{Digite \textbf{Economia de dados} e toque na opção que aparecer com este nome.}
\passo{BraveOrange}{3}{Ative a chave \textbf{Usar o recurso Economia de dados}. Isso bloqueia o gasto invisível de vários apps de uma vez.}
